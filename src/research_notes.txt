### 3.1 Pull the latest model cards from Hugging Face (to see what’s hot)

```python
import requests, json, pandas as pd

HF_API = "https://huggingface.co/api/models"
params = {
    "filter": "language:en,sort:downloads,limit:20",  # top‑20 most‑downloaded English models
}
r = requests.get(HF_API, params=params)
models = r.json()

df = pd.DataFrame([
    {
        "model_id": m["modelId"],
        "downloads": m["downloads"],
        "pipeline_tag": m.get("pipeline_tag"),
        "tags": m.get("tags", []),
        "lastModified": m["lastModified"],
    }
    for m in models
])
print(df.head())
```

**Why:** Quickly surf the community pulse – you’ll see spikes in “llama‑2‑70b”, “gemma‑2b‑it”, “stable‑diffusion‑3”, etc.

### 3.2 Load a state‑of‑the‑art multimodal model and run inference (single‑GPU)

```python
from transformers import AutoModelForCausalLM, AutoProcessor, BitsAndBytesConfig
import torch

model_name = "google/gemma-2b-it"          # replace with any multimodal LLM, e.g., "mistral/mistral‑7b‑instruct"
quant_cfg = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_compute_dtype=torch.bfloat16,
)

model = AutoModelForCausalLM.from_pretrained(
    model_name,
    device_map="auto",
    quantization_config=quant_cfg,
    trust_remote_code=True,   # many community models need this
)

processor = AutoProcessor.from_pretrained(model_name)

prompt = "Explain the difference between LoRA and IA³ in < 2 sentences."
inputs = processor(prompt, return_tensors="pt").to(model.device)

with torch.cuda.amp.autocast():
    output = model.generate(**inputs, max_new_tokens=100, do_sample=False)

print(processor.decode(output[0], skip_special_tokens=True))
```

**Why:** Shows 4‑bit quant + `torch.compile`‑ready inference – the current de‑facto pattern for research‑scale LLMs.

### 3.3 Retrieval‑Augmented Generation (RAG) with LangChain + FAISS

```python
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain.llms import OpenAI
from langchain.chains import RetrievalQA

# 1️⃣ Load documents (could be scraped papers, blog posts, etc.)
docs = ["..."]   # list of strings or TextLoader objects

# 2️⃣ Embed & index
embeddings = OpenAIEmbeddings(model="text-embedding-3-large")
vectorstore = FAISS.from_texts(docs, embedding=embeddings)

# 3️⃣ Build a QA chain
qa = RetrievalQA.from_chain_type(
    llm=OpenAI(model="gpt-4o-mini"),
    retriever=vectorstore.as_retriever(search_kwargs={"k": 4}),
)

# 4️⃣ Ask a trend question
question = "What are the latest advances in parameter‑efficient fine‑tuning as of Q3 2024?"
print(qa.run(question))
```

**Why:** RAG is the primary way to keep LLM answers *fresh* without re‑training the model.

### 3.4 LoRA fine‑tune a 7‑B LLM on a custom dataset (single‑GPU)

```python
!pip install peft transformers datasets bitsandbytes accelerate -q

import torch
from datasets import load_dataset
from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import LoraConfig, get_peft_model, prepare_model_for_int8_training

model_name = "meta-llama/Meta-Llama-3-8B"
tokenizer = AutoTokenizer.from_pretrained(model_name)

# Load a tiny instruction dataset (e.g., Alpaca‑Gpt4)
dataset = load_dataset("tatsu-lab/alpaca", split="train[:1%]")

# 4‑bit int8 + LoRA
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    load_in_4bit=True,
    device_map="auto",
    quantization_config=BitsAndBytesConfig(load_in_4bit=True)
)
model = prepare_model_for_int8_training(model)

lora_cfg = LoraConfig(
    r=64,
    lora_alpha=16,
    target_modules=["q_proj", "v_proj"],
    lora_dropout=0.05,
    bias="none",
    task_type="CAUSAL_LM",
)
model = get_peft_model(model, lora_cfg)

# Simple trainer loop (no HF Trainer for brevity)
optimizer = torch.optim.AdamW(model.parameters(), lr=2e-4)

for epoch in range(2):
    for sample in dataset.shuffle().select(range(200)):
        inputs = tokenizer(sample["instruction"] + sample["input"], return_tensors="pt").to(model.device)
        labels = tokenizer(sample["output"], return_tensors="pt")["input_ids"].to(model.device)
        outputs = model(**inputs, labels=labels)
        loss = outputs.loss
        loss.backward()
        optimizer.step()
        optimizer.zero_grad()
    print(f"epoch {epoch} loss {loss.item():.4f}")
```

**Why:** Shows the **PEFT** workflow that powers most private‑data custom LLMs today.
